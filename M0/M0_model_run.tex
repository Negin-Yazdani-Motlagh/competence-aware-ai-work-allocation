\noindent
This is the M0 model that was run: the original HCOMP specification
with one competence state and three actions.

\section{State, task, and action}
\[
c_t\in[0,1],\qquad
z_t\sim\mathrm{Unif}[0,1],\qquad
a_t\in\{H,L,A\}.
\]
Initial competence: \(c_1=c_0=0.35\).
Horizon: \(T=400\).
Number of runs: \(N=20\), master seed \(42\).

\section{Current-task quality}
\[
\sigma(x)=\frac{1}{1+e^{-x}}.
\]
\begin{align}
p_H(c_t,z_t)
&=\sigma(\alpha_0+\alpha_c c_t-\alpha_z z_t),
\label{eq:pH}\\
p_L(z_t)
&=\sigma(\beta_0-\beta_z z_t),
\label{eq:pL}\\
p_A(c_t,z_t)
&=p_L(z_t)+\bigl[1-p_L(z_t)\bigr]c_t\,p_H(c_t,z_t).
\label{eq:pA}
\end{align}
Quality used in the run:
\[
p(a_t,c_t,z_t)
=
\begin{cases}
p_H(c_t,z_t) & a_t=H,\\
p_L(z_t) & a_t=L,\\
p_A(c_t,z_t) & a_t=A.
\end{cases}
\]

\section{Payoff}
\[
u(a_t,c_t,z_t)=p(a_t,c_t,z_t)-\kappa_{a_t}.
\label{eq:u}
\]

\section{Competence transition}
\[
c_{t+1}
=\Pi_{[0,1]}\bigl[c_t+\rho(m_{a_t}-c_t)\bigr],
\label{eq:trans}
\]
where \(\Pi_{[0,1]}\) clips to \([0,1]\) and \(\rho_a=\rho\) for all actions.

\section{Policies}
\paragraph{H-only.} \(a_t=H\) for all \(t\).
\paragraph{L-only.} \(a_t=L\) for all \(t\).
\paragraph{A-only.} \(a_t=A\) for all \(t\).
\paragraph{Myopic.}
\[
a_t=\arg\max_{a\in\{H,L,A\}} u(a,c_t,z_t).
\]
\paragraph{Dynamic.}
Finite-horizon backward induction. Let \(z\) be discretized on the \(z\)-grid.
\[
V_{T+1}(c)=Bc,
\]
\[
V_t(c)
=
\mathbb{E}_z
\left[
\max_{a\in\{H,L,A\}}
\bigl\{
u(a,c,z)+\gamma V_{t+1}(c'(a,c))
\bigr\}
\right],
\]
\[
c'(a,c)=\Pi_{[0,1]}\bigl[c+\rho(m_a-c)\bigr].
\]
The implemented policy is
\[
a_t
=
\arg\max_{a\in\{H,L,A\}}
\bigl\{
u(a,c_t,z_t)+\gamma V_{t+1}(c'(a,c_t))
\bigr\},
\]
with \(V_{t+1}\) interpolated on the \(c\)-grid.

\section{Period-\(t\) update}
Given \((c_t,z_t)\) and a policy,
\begin{enumerate}
  \item choose \(a_t\);
  \item record \(p_t=p(a_t,c_t,z_t)\) and \(u_t=u(a_t,c_t,z_t)\);
  \item set \(c_{t+1}\) by~\eqref{eq:trans}.
\end{enumerate}
Reported cumulative utility on a run is \(\sum_{t=1}^{T}u_t\).
